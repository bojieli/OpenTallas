from pathlib import Path
import subprocess,json,re,itertools
w=Path('/tmp/pq-return-delay-sweep');w.mkdir(exist_ok=True)
root=Path('/home/ubuntu/OpenTallas');lib=root/'results/uarch/topk_station_SSFF_cell_model_20261002/inputs'; rows=[]
for n in (0,3,4):
 v=f'module top(input clk,input din,output q);\nwire [{n}:0] w;\nBUFx2_ASAP7_75t_R original_input(.A(din),.Y(w[0]));\n'
 for k in range(n):v+=f'HB4xp67_ASAP7_75t_R h{k}(.A(w[{k}]),.Y(w[{k+1}]));\n'
 v+=f'DFFHQNx1_ASAP7_75t_R ff(.CLK(clk),.D(w[{n}]),.QN(q));\nendmodule\n';(w/f'chain{n}.v').write_text(v)
 for corner,slew,cap in itertools.product(('SS','FF'),(0,5,20),(0,.2,.78)):
  t='read_lef /tmp/bf_phase_sta_fixture/asap7_tech_1x_201209.lef\nread_lef /tmp/bf_phase_sta_fixture/asap7sc7p5t_28_R_1x_220121a.lef\n'
  t+=f'read_liberty {lib}/asap7sc7p5t_INVBUF_RVT_{corner}_nldm_220122.lib.gz\nread_liberty {lib}/asap7sc7p5t_SEQ_RVT_{corner}_nldm_220123.lib\nread_verilog {w}/chain{n}.v\nlink_design top\ncreate_clock -name clk -period 833.333 [get_ports clk]\nset_input_delay 0 -clock clk [get_ports din]\nset_input_transition {slew} [get_ports din]\nset_load {cap} [get_nets {{w*}}]\nreport_checks -from [get_ports din] -to [get_pins ff/D] -path_delay min_max -format full_clock_expanded -digits 6\nexit\n'
  name=f'n{n}_{corner}_slew{slew}_cap{cap}';p=w/(name+'.tcl');p.write_text(t);r=subprocess.run(['openroad','-no_init','-exit',str(p)],text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT);(w/(name+'.log')).write_text(r.stdout)
  assert '[ERROR' not in r.stdout and '[WARNING' not in r.stdout,r.stdout
  times=[float(x) for x in re.findall(r'([-0-9.]+)\s+data arrival time',r.stdout) if float(x)>0];assert len(times)==2
  rows.append(dict(chain_cells=n,corner=corner,input_slew_ps=slew,per_net_wire_cap_fF=cap,min_ps=min(times),max_ps=max(times)))
for r in rows:
 b=next(x for x in rows if x['chain_cells']==0 and all(x[k]==r[k] for k in ('corner','input_slew_ps','per_net_wire_cap_fF')))
 r['min_delta_ps']=r['min_ps']-b['min_ps'];r['max_delta_ps']=r['max_ps']-b['max_ps']
(w/'result.json').write_text(json.dumps(rows,indent=2)+'\n')
for n in (3,4):
 a=[r for r in rows if r['chain_cells']==n];print(n,'FF min gain',min(r['min_delta_ps'] for r in a if r['corner']=='FF'),'SS maxcost',max(r['max_delta_ps'] for r in a if r['corner']=='SS'))
