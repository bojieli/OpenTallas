def groups(text,kind):
    for m in re.finditer(r'\b'+kind+r'\s*\(([^)]*)\)\s*\{',text):
        i=m.end();j=i;d=1
        while d: d+=(text[j]=='{')-(text[j]=='}');j+=1
        yield m[1].strip(' "'),text[i:j-1]
def table(t,name):
    choices=list(groups(t,name))
    if len(choices)!=1:raise ValueError('table absent/nonunique')
    t=choices[0][1]
    axes=[[float(x) for x in re.search(r'index_'+str(i)+r'\s*\("([^"]+)"\)',t)[1].split(',')] for i in (1,2)]
    rows=[[float(x) for x in r.split(',')] for r in re.findall(r'"([^"]+)"',re.search(r'values\s*\((.*?)\)\s*;',t,re.S)[1])]
    if len(rows)!=len(axes[0]) or any(len(r)!=len(axes[1]) for r in rows):raise ValueError('table shape')
    return dict(slew=axes[0],load=axes[1],values=rows)
def arcs(body,kind,keys):
    found=[]
    for _,t in groups(body,'timing'):
        if not re.search(r'timing_type\s*:\s*'+kind+r'\s*;',t):continue
        for key in keys:found.append((key,table(t,key)))
    if not found:raise ValueError('source arc missing '+kind)
    return found

def upper(t,slew,load):
    if slew<t['slew'][0] or load<t['load'][0]:raise ValueError('subgrid domain')
    rows=[i for i,x in enumerate(t['slew']) if x>=slew];cols=[i for i,x in enumerate(t['load']) if x>=load]
    if not rows or not cols:raise ValueError('characterized domain exceeded')
    # Monotonicity is not assumed: bound all points through rounded ceiling.
    return max(t['values'][i][j] for i in range(rows[0]+1) for j in range(cols[0]+1))
def minimum(t): return min(x for r in t['values'] for x in r)
def library():
    raw=json.loads((C.OUT/'inputs/cell_bodies.json').read_text());facts,_=C.facts(); result={}
    for corner,cells in raw.items():
        result[corner]={}
        for name in (C.HQ,C.ASR,C.NAND,C.INV,C.BUF):
            body=cells[name];isff=name in (C.HQ,C.ASR)
            result[corner][name]=dict(tables=arcs(body,'rising_edge' if isff else 'combinational',('cell_rise','cell_fall','rise_transition','fall_transition')),pins=facts[name][corner]['pins'])
            if isff:
                result[corner][name]['setup']=arcs(C.group(body,'pin','D'),'setup_rising',('rise_constraint','fall_constraint'))
                result[corner][name]['hold']=arcs(C.group(body,'pin','D'),'hold_rising',('rise_constraint','fall_constraint'))
    return result

def RC():
    text=(OUT/'inputs/setRC.tcl').read_text()
    rc={layer:(float(r),float(c)) for layer,r,c in re.findall(r'set_layer_rc -layer (M[2345]) -resistance ([\d.Ee+-]+) -capacitance ([\d.Ee+-]+)',text)}
    if len(rc)!=4:raise ValueError('routing source')
    return dict(R_kohm_per_um=max(v[0] for v in rc.values()),C_fF_per_um=max(v[1] for v in rc.values()),two_via_R_kohm=2*.0172,source_layers=rc)

def evaluate(corner,launch,nand_count,global_buffers=0,local_buffers=0,selector=False,selector_local_nodes=1):
    lib=library()[corner];rc=RC();steps=[];elapsed=0; slew=320.0 # declared clock ingress ceiling
    def step(name,cap,sink,role):
        nonlocal elapsed,slew
        cell=lib[name]
        if cap<.72 or cap>cell['pins']['QN' if name in (C.HQ,C.ASR) else 'Y']['max_cap_fF']:raise ValueError('cap domain')
        delay=max(upper(t,slew,cap) for k,t in cell['tables'] if k.startswith('cell_'))
        transition=max(upper(t,slew,cap) for k,t in cell['tables'] if k.endswith('transition'))
        wire=cap-sink
        if wire<=0:raise ValueError('no positive wire budget')
        length=wire/rc['C_fF_per_um'];wr=(length*rc['R_kohm_per_um']+rc['two_via_R_kohm'])*cap
        out=transition+2.2*wr;elapsed+=delay+wr
        steps.append(dict(role=role,master=name,input_slew_ps=slew,total_load_fF=cap,sink_cap_fF=sink,
                          wire_cap_fF=wire,wire_length_ceiling_um=length,wire_delay_upper_ps=wr,
                          cell_delay_upper_ps=delay,cell_output_slew_ps=transition,receiver_slew_upper_ps=out,cumulative_ps=elapsed))
        if out>320:raise ValueError('receiver slew exceeds320ps')
        slew=max(5.0,out)
    icap=lib[C.INV]['pins']['A']['cap_fF'];bcap=lib[C.BUF]['pins']['A']['cap_fF']
    ncap=max(lib[C.NAND]['pins'][p]['cap_fF'] for p in ('A','B'));dcap=max(lib[n]['pins']['D']['cap_fF'] for n in (C.HQ,C.ASR))
    step(launch,1.44,icap,'launch QN->restoringINV')
    if selector:
        # Clocked selector -> global fanout tree -> node-local INV ->69bit
        # vector select tree. Both forests are already priced b991 sections.
        step(C.INV,1.44 if global_buffers or selector_local_nodes==1 else 2.88,bcap if global_buffers else selector_local_nodes*icap,'restore selector QN')
        for j in range(global_buffers):step(C.BUF,11.52,8*bcap if j+1<global_buffers else 8*icap,'global selector fanout')
        step(C.INV,1.44,bcap,'node-local shared select inversion')
        for j in range(local_buffers):step(C.BUF,23.04 if j+1==local_buffers else 11.52,8*ncap if j+1==local_buffers else 8*bcap,'local69bit selector fanout')
    else:step(C.INV,1.44,ncap,'restore payload QN')
    for j in range(nand_count):step(C.NAND,1.44,dcap if j+1==nand_count else ncap,'mux NAND '+str(j+1))
    # Constraint table index1=data transition,index2=clock transition.
    setup=max(upper(t,slew,320) for n in (C.HQ,C.ASR) for _,t in lib[n]['setup'])
    total=elapsed+setup+60+25
    return dict(corner=corner,steps=steps,destination_slew_ps=slew,setup_ps=setup,
                setup_uncertainty_ps=60,adverse_skew_budget_ps=25,path_and_constraints_ps=total,
                period_ps=1000/1.2,remaining_ps=1000/1.2-total,conditional_setup_screen_PASS=total<=1000/1.2,
                physical_routes_proven=False)

def early(launch,nands,buffers,selector):
    lib=library()['FF'];seq=[launch,C.INV]+([C.BUF]*buffers+[C.INV] if selector else [])+[C.NAND]*nands
    total=sum(min(minimum(t) for k,t in lib[n]['tables'] if k.startswith('cell_')) for n in seq)
    hold=max(max(x for r in t['values'] for x in r) for n in (C.HQ,C.ASR) for _,t in lib[n]['hold'])
    return dict(source_minimum_cell_sum_ps=total,destination_hold_upper_ps=hold,hold_uncertainty_ps=25,
                adverse_skew_ps=25,conditional_zero_wire_margin_ps=total-hold-50,
                characterized_minimum_only=True,subgrid_load_or_slew_not_covered=True,physical_FF_hold=False)

def model():
    # Exactly ONE proposal, cut after logical level5: local3 + root2.
    # 32 intermediate69bit streams then final69bit stream; no variant sweep.
    data_bits=(32+1)*69; control_bits=(5+16+1)+(16+1) # stage selector/expectedrow/valid
    timing={}
    specs={'stage1_payload':(C.HQ,10,0,0,False,1),
           'stage1_bank_local_position_bit0':(C.ASR,10,0,3,True,3),
           'stage1_shared_root_bank_bit0':(C.ASR,4,2,3,True,1),
           'stage1_shared_root_bank_bit1':(C.ASR,2,2,3,True,1),
           'stage2_payload':(C.HQ,10,0,0,False,1),
           'stage2_shared_root_bank_bit2':(C.ASR,10,2,3,True,1),
           'stage2_shared_root_bank_bit6':(C.ASR,2,0,3,True,1)}
    domain_failure={};ff_paths={}
    for name,spec in specs.items():
        for corner,dest in [('SS',timing),('FF',ff_paths)]:
            try:dest[name]=evaluate(corner,*spec)
            except ValueError as e:domain_failure[corner+'.'+name]=str(e)
    accepted=not domain_failure and all(v['conditional_setup_screen_PASS'] for v in timing.values())
    f,_=C.facts();hq=f[C.HQ]['SS'];asr=f[C.ASR]['SS'];inv=f[C.INV]['SS'];nand=f[C.NAND]['SS'];buf=f[C.BUF]['SS']
    # Stalling drain must hold all pipeline states; raw field capture is never
    # backpressured. MUX holds cost3NAND/bit, share vector selects, then freeze.
    states=data_bits+control_bits;hold_mux=3*states;restore=states
    clk_delta=sum(C.tree(39744+1483+states))-sum(C.tree(39744+1483))
    rst_delta=sum(C.tree(1483+control_bits))-sum(C.tree(1483))
    vector_hold=(33+2)*sum(C.tree(69));root_select_BUF=sum(sum(C.tree(n)) for n in (64,32,16,8,4,2,1));scalar_hold=states # conservative per-bit inversion bound explicitly priced
    body=data_bits*hq['area_um2']+control_bits*asr['area_um2']+restore*inv['area_um2']+hold_mux*nand['area_um2']+scalar_hold*inv['area_um2']+(clk_delta+rst_delta+vector_hold+root_select_BUF)*buf['area_um2']+control_bits*.04374
    ff={'stage1_short_payload_4mux':early(C.HQ,8,0,False),
        'stage2_payload_5mux':early(C.HQ,10,0,False),
        'short_selector_final_mux':early(C.ASR,2,3,True)}
    return dict(schema=1,verdict='PASS_CONDITIONAL_CELL_SCREEN_ONLY' if accepted else 'FAIL_ONE_BALANCED_CUT_CONDITIONAL_SS',
        proposal=dict(cut_after_mux_level=5,stages=2,maximum_mux_levels_per_stage=5,
                      intermediate_streams=32,raw_width=69,final_streams=1,one_proposal_no_sweep=True),
        selector_binding=dict(bank_local_read_index_bits=384,local_level1_nodes_per_bank_max=3,shared_bankID_root_level_node_counts=[64,32,16,8,4,2,1],first_cut_consumes_bankID_bits=[0,1],remaining_after_cut_bankID_bits=[2,3,4,5,6],global_bankID_fanout_tree_added_not_free=True),
        source_unit_ps_and_fF='pinned ASAP7 source library interiors and raw source hashes from b991; no extrapolation',
        SS_paths=timing,FF_propagated_paths=ff_paths,characterization_domain_failures=domain_failure,FF_hold_screens=ff,
        SS_setup_closed=False,FF_hold_closed=False,physical_slot_fit=False,
        cost=dict(data_FF_bits=data_bits,async_control_FF_bits=control_bits,restoring_INV_cells=restore,
                  hold_mux_NAND2_cells=hold_mux,hold_mux_INV_upper_cells=scalar_hold,
                  extra_clock_BUF=clk_delta,extra_reset_BUF=rst_delta,vector_hold_BUF=vector_hold,
                  extra_TIEHI_cells=control_bits,extra_global_root_select_BUF=root_select_BUF,cell_body_delta_um2=body,
                  reservation_delta_mm2_at50pct=body*2/1e6,
                  extra_clock_pin_SS_fF=data_bits*hq['pins']['CLK']['cap_fF']+control_bits*asr['pins']['CLK']['cap_fF'],
                  extra_reset_pin_SS_fF=control_bits*asr['pins']['RESETN']['cap_fF']),
        RC=RC(),
        conditional_calendar=dict(read_request_first=423,read_request_last=998,intermediate_capture_first=423,
            final_output_postNBA_first=424,downstream_accept_first=425,downstream_accept_last=1000,
            selector_preparation='bank0/position0 context is reserved beforeGO; scalar selectors stable before stage1 acceptance; subsequent selectors advance on acceptance',
            pipeline_stages=2,maximum_scalar_requests_per_edge=1,maximum_outstanding_read_requests=2,
            calendar_is_conditional_not_admitted=not accepted,full_program_consumer_deadline=None),
        backpressure='Only scalar drain may freeze when downstream refuses; two accepted request credits retained until delivery; all pipeline cells stall together. No field-ready or global ECC ACK.',
        lease='Frozen generation/user/context remains live until sourceidle +576 captures +pipeline/packet empty +delivery fence +causal home visibility +provenance; original roots have no epoch bits',
        context_control='expected row16 and valid1 carried through both stages; upper bank selector5 carried after cut. Command identity frozen, never assigned to raw root tag.',
        timeout=dict(default1024_FAIL_preserved=True,proposed4096_selected=False),
        unknowns=['actual placement/RC/clock skew and min-load hold geometry','global selector branching source capture timing','VM bridge/service consumer deadline and causal fences'],
        RTL_or_build_admitted=False,whole_token_or_remote_provider_credit=False)

def pins():
    ps=[Path(__file__),ROOT/'tools/dsrom_I66_capture_cells.py',C.OUT/'model.json',C.OUT/'inputs/cell_bodies.json',OUT/'inputs/setRC.tcl']
    return {str(p.relative_to(ROOT)):hashlib.sha256(p.read_bytes()).hexdigest() for p in ps}
if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--out',type=Path,required=True);p.add_argument('--verify',action='store_true');a=p.parse_args()
    m=model();m['source_pins']=pins();data=(json.dumps(m,sort_keys=True,indent=2)+'\n').encode()
    if a.verify:
        if a.out.read_bytes()!=data:raise SystemExit('FAIL replay')
