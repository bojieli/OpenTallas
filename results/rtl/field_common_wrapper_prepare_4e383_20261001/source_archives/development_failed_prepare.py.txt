#!/usr/bin/env python3
"""Prepare only: no compiler/simulator entry point or implicit execution approval."""
import argparse
import hashlib
import json
from pathlib import Path
import re
import subprocess

ROOT = Path(__file__).resolve().parents[1]
RECORD = ROOT / 'results/rtl/field_common_wrapper_prepare_4e383_20261001'
SOURCE = '4e38326d6f361bc85e660f48c59c355e2bb95274'
SOURCES = ['rtl/common/ot_prefix.sv', 'rtl/hdc/ot_hdc_delay.sv'] + [
    'rtl/v41rom/ot_v41_' + n + '.sv' for n in
    ('fadd', 'bmul2', 'chain2', 'segtree2', 'bf16_lanes2')]
TOPS = ('chain16', 'chain8', 'tree16', 'bf16')


def sha(data):
    return hashlib.sha256(data).hexdigest()


def read_sources(repo=ROOT):
    pins = json.loads((RECORD / 'resource_estimate.json').read_text())['source_pins']
    result = {}
    for path in SOURCES:
        data = subprocess.check_output(['git', 'show', SOURCE + ':' + path], cwd=repo)
        if sha(data) != pins[path]['sha256']:
            raise ValueError('source pin mismatch: ' + path)
        result[path] = data.decode()
    return result


def replace_helpers(text):
    for name, body in [('ot_v41_ksadd',
            "\n    assign {cout,s} = {1'b0,a} + {1'b0,b} + {{W{1'b0}},cin};\n"),
            ('ot_v41_inc', "\n    assign {co,y} = {1'b0,a} + {{W{1'b0}},inc};\n")]:
        matches = list(re.finditer(r'\bmodule\s+' + name + r'\b.*?\bendmodule\b', text, re.S))
        if len(matches) != 1:
            raise ValueError('helper declaration count: ' + name)
        match = matches[0]
        signature_end = text.index(');', match.start()) + 2
        body_end = text.index('endmodule', signature_end)
        text = text[:signature_end] + body + text[body_end:]
    return text


def namespace(text, names, prefix):
    # Match whole identifiers, never arithmetic, pipeline cuts or reset statements.
    return re.sub(r'\b(?:' + '|'.join(map(re.escape, sorted(names))) + r')\b',
                  lambda m: prefix + m[0], text)


def generate(sources, negative_control=False):
    names = set(re.findall(r'\bmodule\s+(\w+)', '\n'.join(sources.values())))
    if names != {'ot_v41_ksadd', 'ot_v41_inc', 'ot_hdc_delay', 'ot_v41_fadd',
                 'ot_v41_cut', 'ot_v41_bmul2', 'ot_v41_chain2',
                 'ot_v41_segtree2', 'ot_v41_bf16_lanes2'}:
        raise ValueError('unexpected dependency module set')
    files = {}
    for side in ('ref_', 'cand_'):
        for path, text in sources.items():
            if side == 'cand_' and path == SOURCES[0]:
                text = replace_helpers(text)
                if negative_control:
                    text = text.replace("+ {{W{1'b0}},cin};",
                                        "+ {{W{1'b0}},cin}) ^ {{W{1'b0}},1'b1};")
                    text = text.replace("= {1'b0,a} + {1'b0,b}",
                                        "= ({1'b0,a} + {1'b0,b}")
            files[side + Path(path).name] = namespace(text, names, side)
    return files


def commands(top):
    if top not in TOPS:
        raise ValueError('unknown top')
    # A cgroup is mandatory; RLIMIT_AS alone does not cap aggregate compiler children.
    cap = ['systemd-run', '--user', '--wait', '--pipe', '--collect',
           '-p', 'MemoryMax=4294967296', '-p', 'MemorySwapMax=0',
           '-p', 'CPUQuota=200%', '-p', 'TasksMax=64']
    files = [s + Path(p).name for s in ('ref_', 'cand_') for p in SOURCES]
    return {'compile': cap + ['timeout', '--kill-after=5s', '600s', 'verilator',
        '--binary', '--timing', '--build', '-j', '2', '-Wno-fatal',
        '--top-module', 'tb_' + top, '--Mdir', 'obj_' + top,
        'tb_field_common_wrappers.sv'] + files,
        'simulate': cap + ['timeout', '--kill-after=5s', '60s', './obj_' + top + '/Vtb_' + top]}


def prepare(out, negative_control=False):
    # Never overwrite receipts or even partially reuse an earlier attempt.
    if out.exists():
        raise FileExistsError(out)
    sources = read_sources()
    generated = generate(sources, negative_control)
    generated['tb_field_common_wrappers.sv'] = (ROOT / 'rtl/test/tb_field_common_wrappers.sv').read_text()
    out.mkdir(parents=True)
    for name, content in generated.items():
        (out / name).write_text(content)
    record = {'status': 'PREPARED_NOT_EXECUTED', 'source_commit': SOURCE,
              'resource_estimate_sha256': sha((RECORD / 'resource_estimate.json').read_bytes()),
              'files_sha256': {n: sha((out/n).read_bytes()) for n in generated},
              'negative_control': negative_control,
              'commands_proposed_only': {t: commands(t) for t in (('chain8',) if negative_control else TOPS)},
              'execution_gate': 'BLOCKED: parent-reviewed Peirce sequential PASS and explicit GO required',
              'future_runner_requirements': ['verify all hashes before execution',
                'run tops serially, cap whole process tree, no uncapped fallback or retry',
                'preserve compiler/simulator versions, stdout/stderr and cgroup memory.peak',
                'require PASS marker and all runtime coverage assertions for each top',
                'separate fresh --negative-control attempt: flip actual common adder sum bit zero; require comparison FAIL (not just any nonzero exit)',
                'retain every failed attempt; no hardware/adoption credit']}
    (out/'preparation.json').write_text(json.dumps(record, indent=2, sort_keys=True)+'\n')
    return record


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--negative-control', action='store_true')
    parser.add_argument('--out', type=Path, required=True)
    args = parser.parse_args()
    prepare(args.out, args.negative_control)
